\subsection{Funding Rate Signal Generation & Ranking}
\begin{algorithm}
\caption{Signal Generation and Utility Ranking}
\begin{algorithmic}[1]
\FOR each asset
    \STATE Calculate basis $B^X_t$, volatility $\sigma_{B^X,t}$ (rolling window).
    \STATE Estimate $\mathbb{E}_t[FR^X]$ (e.g., simple avg).
    \STATE Calculate $\mathbb{E}_t[\pi^X_t]$ (incl. costs $C^X_t$ + slippage est.).
    \STATE Calculate $\mathbb{E}_t[\pi^{AB}_t]$ (cost-adjusted NFD).
    \STATE IF profitable:
        \STATE Calc Kelly size $f^*_{actual}$ and $f_{constrained}$.
        \STATE \textbf{Calculate Utility $U = \pi_{adj} - \lambda \sigma_{B,t}^2$.}
        \STATE Add (Asset, ..., Size $f_{constrained}$, Utility $U$) to OpportunityList.
    \ENDIF
\ENDFOR
\STATE \textbf{Rank OpportunityList by descending Utility $U$.}
\RETURN Ranked OpportunityList
\end{algorithmic}
\end{algorithm}

\subsection{Risk Management and Sizing}
\begin{algorithm}
\caption{Risk Management and Sizing (Dynamic Limits)}
\begin{algorithmic}[1]
\REQUIRE Ranked OpportunityList, $w_{current}$, $\Sigma$ (updated periodically), Risk Limits ($VaR_0$, etc.)
\ENSURE Target Portfolio $w_{target}$
\STATE Calculate $\sigma_{P, current} = \sqrt{w_{current}^T \Sigma w_{current}}$. Calculate current CVaR.
\STATE \textbf{Calculate current market vol $\sigma_{mkt,t}$ (e.g., 1h rolling).}
\STATE \textbf{Calculate dynamic risk limit $VaR_t = VaR_0 \times (\sigma_{mkt,t} / \sigma_{mkt,0})$.}
\STATE IF current VaR > $VaR_t$ OR other limits breached: \textbf{Reduce Risk}.
\STATE Initialize $w_{proposed} = w_{current}$.
\FOR each ranked opportunity $O$ (by Utility)
    \STATE Calculate hypothetical portfolio $w_{hypothetical}$ using $f_{constrained}$ size.
    \STATE Check if $\sigma_{P, hypothetical}$ violates limits.
    \STATE IF within limits: $w_{proposed} = w_{hypothetical}$. ELSE: Skip/Reduce.
\ENDFOR
\STATE $w_{target} = w_{proposed}$.
\STATE Trigger Collateral Check based on $w_{target}$.
\RETURN $w_{target}$
\end{algorithmic}
\end{algorithm}

\subsection{Execution Handling}
\begin{algorithm}
\caption{Execution Handling (Utility Ranked)}
\begin{algorithmic}[1]
\REQUIRE $w_{target}$, $w_{current}$, Market Data
\ENSURE Orders Placed/Managed
\STATE Calculate $\Delta w = w_{target} - w_{current}$.
\STATE Model latency $t_{exec}$.
\FOR each required trade in $\Delta w$ (\textbf{ordered by Utility rank})
    \STATE Execute Cross-Exchange or Single-Exchange (handle ED25519, use FOK/IOC, handle partials, adjust for slippage estimate).
    \STATE Monitor order status.
    \STATE Update internal position state.
    \STATE \textbf{Apply interim scaling during transfers if needed.}
\ENDFOR
\end{algorithmic}
\end{algorithm}
